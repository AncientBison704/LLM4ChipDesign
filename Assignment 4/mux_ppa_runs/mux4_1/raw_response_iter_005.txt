module mux4_1(input [1:0] sel, input [3:0] in, output out);
    wire sel0_n, sel1_n;
    wire a, b;
    
    // Decode selection
    assign sel0_n = ~sel[0];
    assign sel1_n = ~sel[1];
    
    // Mux logic
    assign a = (sel[0] & in[1]) | (sel0_n & in[0]);
    assign b = (sel[0] & in[3]) | (sel0_n & in[2]);
    assign out = (sel[1] & b) | (sel1_n & a);
endmodule
